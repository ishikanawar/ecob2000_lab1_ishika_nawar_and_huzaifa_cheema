# ---- Setup ----
n <- 100
p <- 1/6

# ---- Step 1: Mean and SD of the null distribution ----
mu <- n * p
sigma <- sqrt(n * p * (1 - p))
cat("Mean:", mu, " SD:", sigma, "\n")

# ---- Step 2: Normal approximation cutoffs (z = 1.96 for alpha = 0.05, two-tailed) ----
z <- qnorm(0.975)  # 1.96
lower_approx <- mu - z * sigma
upper_approx <- mu + z * sigma
cat("Normal approx cutoffs:", lower_approx, upper_approx, "\n")

n <- 100      # number of rolls
p <- 1/6      # null hypothesis: probability of rolling a 6 on a fair die

# Rejection region: reject H0 (call the die unfair) if X <= 9 or X >= 25

# Lower tail: P(X <= 9)
lower_tail <- pbinom(9, size = n, prob = p)
cat("P(X <= 9)  =", lower_tail, "\n")
P(X <= 9)  = 0.02129241 
# Upper tail: P(X >= 25)  =  1 - P(X <= 24)
upper_tail <- 1 - pbinom(24, size = n, prob = p)
cat("P(X >= 25) =", upper_tail, "\n")
P(X >= 25) = 0.02170338 
# Type I error rate = sum of both tails (mutually exclusive events)
alpha_actual <- lower_tail + upper_tail
cat("Actual Type I error rate =", alpha_actual, "\n")
Actual Type I error rate = 0.04299579
